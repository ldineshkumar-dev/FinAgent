-- This is a complete and accurate schema of the database.
-- It includes all tables and columns to ensure the AI can generate correct queries.
-- Foreign key relationships are defined to guide the AI in constructing complex joins.

CREATE TABLE i_acct_account_type_mst (
    id INT PRIMARY KEY,

    modify_ip VARCHAR(255),
    modify_dnt DATETIME,

    atm_parent_type INT,

    id INT PRIMARY KEY,
    comp_id INT,

    create_by INT,
    create_ip VARCHAR(255),
    create_dnt DATETIME,

    modify_ip VARCHAR(255),
    modify_dnt DATETIME,
    agm_name VARCHAR(255),
    agm_parent_group_id INT,
    agm_atm_id INT,
    agm_balance_sheet_name VARCHAR(255),
    agm_include_in VARCHAR(255),
    agm_is_system BOOLEAN,
    agm_sort_order INT,
    agm_ref_id INT,
    agm_ref_type VARCHAR(255),

-- This is a view, not a table. It pre-calculates debit and credit amounts.
CREATE VIEW viewAccountDetail AS
SELECT
    asm_date,
    asm_brm_id,
    asm_acc_id,
    acc_agm_id,
    CASE WHEN asm_acc_type = 'debit' THEN asm_amount ELSE 0 END AS DRAMT,
    CASE WHEN asm_acc_type = 'credit' THEN asm_amount ELSE 0 END AS CRAMT,
    -- Assuming a placeholder for foreign currency conversion
    CASE WHEN asm_acc_type = 'debit' THEN asm_amount ELSE 0 END AS DRAMT_USD,
    CASE WHEN asm_acc_type = 'credit' THEN asm_amount ELSE 0 END AS CRAMT_USD,
    asm_is_reconcile as is_reconcile,
    s.comp_id
FROM i_acct_account_sheet_mst s
LEFT JOIN i_acct_account_mst a ON s.asm_acc_id = a.id;
